cmake_minimum_required(VERSION 3.16)
project(MKWVCAudioRegression LANGUAGES C CXX)
enable_testing()
add_subdirectory(../third_party/rnnoise rnnoise)
add_executable(audio_processor_harness AudioProcessorHarness.cpp ../src/AudioProcessor.cpp)
target_include_directories(audio_processor_harness PRIVATE ../include)
target_compile_features(audio_processor_harness PRIVATE cxx_std_20)
target_link_libraries(audio_processor_harness PRIVATE mkwvc_rnnoise)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(MKWVC_TEST_SPEECH "" CACHE FILEPATH "Optional 16-bit mono speech WAV for the speech-plus-noise regression")
add_test(NAME noise_suppression COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_noise_suppression.py" "$<TARGET_FILE:audio_processor_harness>" "${MKWVC_TEST_SPEECH}")
